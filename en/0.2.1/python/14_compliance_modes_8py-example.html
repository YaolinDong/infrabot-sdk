--mode, --seconds

3# Before enabling the arm, clear its workspace and verify the hardware emergency stop.
4# Local compliance modes run on the MCU at 1 kHz without a PC stream. Push the arm by hand while it runs.
5# STOP at the end brakes it and holds it in IDLE. Gains: see 08_parameters.
6import time
7
8import infrabot
9from example_common import arguments, connected
10
11MODES = {
12 "gravity-free": infrabot.Mode.GravityFree,
13 "gravity-lock": infrabot.Mode.GravityLock,
14 "joint-impedance": infrabot.Mode.JointImpedance,
15 "cartesian-impedance": infrabot.Mode.CartesianImpedance,
16 "joint-admittance": infrabot.Mode.JointAdmittance,
17 "cartesian-admittance": infrabot.Mode.CartesianAdmittance,
18}
19
20
21def extra(parser):
22 parser.add_argument("--mode", choices=MODES, default="cartesian-impedance")
23 parser.add_argument("--seconds", type=float, default=1.0)
24
25
26if __name__ == "__main__":
27 args = arguments(extra=extra)
28 with connected(args, enable_arm=True) as robot:
29 robot.switch_mode(MODES[args.mode])
30 print(f"{args.mode} for {args.seconds:g} s")
31 end = time.monotonic() + args.seconds
32 while time.monotonic() < end:
33 time.sleep(0.5)
34 j = robot.state().joints
35 print(" ".join(f"{q:+.2f}" for q in j.position[:7]))
