3# Read only: prints the controller state, every joint and the link quality for a few seconds, then the
4# fault log. No heartbeat is sent and nothing is enabled. Connecting registers for telemetry, so do not run
5# it on a PC where another program controls the arm (docs/examples.md).
6import time
7
8from example_common import arguments, connected
9
10
11def extra(parser):
12 parser.add_argument("--seconds", type=float, default=1.0)
13
14
15if __name__ == "__main__":
16 args = arguments(extra=extra)
17 with connected(args) as robot:
18 # The first telemetry packet follows the handshake.
19 first = time.monotonic() + 0.5
20 while robot.state().telemetry_age_ms > 100:
21 if time.monotonic() >= first:
22 raise TimeoutError("No telemetry")
23 time.sleep(0.005)
24 end = time.monotonic() + args.seconds
25 while True:
26 s = robot.state()
27 print(f"{s.operational_status.name} mode={s.mode.name} enabled=0x{s.enabled_mask:02x} "
28 f"telemetry age {s.telemetry_age_ms} ms")
29 for i in range(8):
30 j = s.joints
31 print(f" J{i + 1} q={j.position[i]:+.3f} rad dq={j.velocity[i]:+.3f} rad/s "
32 f"tau={j.torque[i]:+.2f} Nm drive={j.drive_state[i]} {j.temperature[i]} C")
33 if time.monotonic() >= end:
34 break
35 time.sleep(0.5)
36 link = robot.link_stats()
37 print(f"link: {link.rx_last_second} packets/s, lost {link.lost_last_second}, "
38 f"longest gap {link.rx_interval_max_us} us")
39 for event in robot.fault_log():
40 print(f"fault {event.kind.name} J{event.joint + 1}: {event.message}")
