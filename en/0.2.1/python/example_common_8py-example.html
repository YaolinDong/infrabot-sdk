11def arguments(*, log=False, extra=None):

24@contextmanager
25def connected(args, *, enable_arm=False):
26 robot = infrabot.Robot(args.ip, args.command_port, args.telemetry_port)
27 # A read-only session sends no heartbeat, so it never looks like the controlling PC.
28 robot.set_heartbeat(enable_arm)
29 robot.connect()
30 try:
31 if not robot.info().received:
32 raise RuntimeError("Controller did not reply")
33 if enable_arm:
34 # DMIO can arrive before the first DMTL. Match the SDK's 100 ms freshness check.
35 deadline = time.monotonic() + 0.5
36 while robot.state().telemetry_age_ms > 100:
37 if time.monotonic() >= deadline:
38 raise TimeoutError("No fresh telemetry before drive setup")
39 time.sleep(0.005)
40 robot.set_heartbeat(True)
41 s = robot.state()
42 arm_enabled = (s.enabled_mask & infrabot.ARM_MASK) == infrabot.ARM_MASK
43 if s.enabled and not (arm_enabled and s.mode == infrabot.Mode.Idle and s.profile_ready
44 and s.profile == infrabot.MotorProfile.Mit):
45 raise RuntimeError("The arm is enabled outside IDLE/MIT or only partly: stop or disable it first")
46 if not arm_enabled:
47 # A previous example left the arm enabled and holding in IDLE: reuse it as it is.
48 robot.set_motor_profile(infrabot.MotorProfile.Mit)
49 robot.servo_on(infrabot.MotorGroup.Arm)
50 # A mode switch needs a still arm. Right after a previous example stopped a local mode, the
51 # hold is still settling for a moment.
52 deadline = time.monotonic() + 3.0
53 while max(abs(v) for v in robot.state().joints.velocity[:7]) > 0.05:
54 if time.monotonic() >= deadline:
55 raise RuntimeError("The arm does not come to rest (above 0.05 rad/s): let go of it")
56 time.sleep(0.01)
57 try:
58 yield robot
59 finally:
60 if enable_arm:
61 robot.stop() # controlled stop to enabled hold
62 finally:
63 robot.disconnect()
