3# Before enabling the arm, clear its workspace and verify the hardware emergency stop.
4# RT Cartesian stream: servep at 100 Hz (the MCU interpolates at 1 kHz and stops after a 100 ms gap).
5# The target rises --dz m along z and comes back on a smooth cosine over --seconds.
6import math
7import time
8
9import infrabot
10from example_common import arguments, connected
11
12
13def extra(parser):
14 parser.add_argument("--dz", type=float, default=0.01)
15 parser.add_argument("--seconds", type=float, default=4.0)
16
17
18if __name__ == "__main__":
19 args = arguments(extra=extra)
20 with connected(args, enable_arm=True) as robot:
21 robot.switch_mode(infrabot.Mode.RtCartesian)
22 start = infrabot.Robot.forward(robot.hold_command()[:7])
23 command = infrabot.CartesianCommand()
24 command.limits.linear_velocity = 0.05
25 command.limits.angular_velocity = 0.3
26 pose = list(start)
27 t0 = time.monotonic()
28 while (t := time.monotonic() - t0) < args.seconds:
29 pose[2] = start[2] + args.dz * 0.5 * (1 - math.cos(2 * math.pi * t / args.seconds))
30 command.pose = pose
31 robot.servep(command)
32 time.sleep(0.01)
33 command.pose = start
34 robot.servep(command)
35 print("stream ended at the start pose")
