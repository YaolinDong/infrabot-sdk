3# Prints one diagnostic snapshot: loop timing, CAN and command counters, memory, drive states. Read only;
4# it also works from a second PC while another controls the arm. The arm is not enabled and does not move.
5import argparse
6
7from example_common import connected
8
9
10def arguments():
11 parser = argparse.ArgumentParser()
12 parser.add_argument("--ip", default="192.168.0.50")
13 parser.add_argument("--command-port", type=int, default=5001)
14 parser.add_argument("--telemetry-port", type=int, default=5000)
15 return parser.parse_args()
16
17
18if __name__ == "__main__":
19 args = arguments()
20 with connected(args) as robot:
21 d = robot.diagnostics()
22 print(f"uptime {d.uptime_ms / 1000:.1f} s, controller status {d.controller_status}")
23 print(f"last second: control {d.control_hz} Hz, missed {d.control_missed}, "
24 f"period max {d.control_period_max_us} us")
25 print(f" CAN tx {d.can_tx} rx {d.can_rx} tx errors {d.can_tx_errors}; "
26 f"commands {d.command_rx}, invalid {d.command_invalid}; telemetry {d.telemetry_tx}")
27 print(f"since boot: missed {d.control_missed_total}, CAN tx errors {d.can_tx_errors_total}, "
28 f"CAN error events {d.can_error_events}, invalid commands {d.command_invalid_total} "
29 f"(last reason {d.invalid_reason})")
30 flags = (" error-passive" if d.can_error_passive else "") + (" BUS-OFF" if d.can_bus_off else "")
31 print(f"CAN TEC {d.can_tec} REC {d.can_rec}{flags}")
32 print(f"heap free {d.heap_free_bytes} B (lowest {d.heap_min_free_bytes} B); least stack free: "
33 f"control {d.stack_free_control_bytes} B, status {d.stack_free_status_bytes} B, "
34 f"ethernet {d.stack_free_ethernet_bytes} B")
35 print("drive states:", " ".join(f"J{i + 1}={s}" for i, s in enumerate(d.drive_state)))
