3# Lists the controller's tunable parameters (gravity, impedance, admittance). The arm is not enabled and
4# does not move. --set NAME VALUE writes one first: RAM only, the controller slews its applied value
5# toward the requested one, and every parameter returns to its default at reboot.
6import argparse
7
8from example_common import connected
9
10
11def arguments():
12 parser = argparse.ArgumentParser()
13 parser.add_argument("--ip", default="192.168.0.50")
14 parser.add_argument("--command-port", type=int, default=5001)
15 parser.add_argument("--telemetry-port", type=int, default=5000)
16 parser.add_argument("--set", nargs=2, action="append", default=[], metavar=("NAME", "VALUE"))
17 return parser.parse_args()
18
19
20if __name__ == "__main__":
21 args = arguments()
22 with connected(args) as robot:
23 for name, value in args.set:
24 written = robot.set_parameter(name, float(value))
25 print(f"set {written.name}: requested {written.requested:g}, applied {written.applied:g}")
26 for p in robot.list_parameters():
27 print(f"{p.name:<20} {p.requested:9.3f} (applied {p.applied:9.3f}) [{p.minimum:g}..{p.maximum:g}] {p.unit}")
